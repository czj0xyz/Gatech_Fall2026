\section{More on Boolean analysis related problems}

\begin{enumerate}
    \item KKL using localization schemes.
    
    WGE: 
        \begin{align}\label{eq:WGE}
            \Gamma_{\theta} U_{\theta \to 1} f \leq e^{-2\rho t(\theta)} U_{\theta \to 1} \Gamma_1 f %\iff \norm{\nabla_w P_t f}^2_2 \leq e^{-\rho_{\mathrm{GE}} t} P_t(\norm{\nabla_w f}^2_2); \tag{WGE}
        \end{align}
        SGE:
        \begin{align}\label{eq:SGE}
            \sqrt{\Gamma_{\theta} U_{\theta \to 1} f} \leq e^{-2\rho t(\theta)} U_{\theta \to 1} \sqrt{\Gamma_1 f} %\iff \norm{\nabla_w P_t f}_2 \leq e^{-\rho_{\mathrm{SGE}} t} P_t(\norm{\nabla_w f}_2); \tag{SGE}
        \end{align}
        
        \begin{align*}
            \Var(f) \le C \sum_i \E[\Var_i(f)] \tag{Poincare} \\
            \sum_i \Var(\E_i[f]]) \le C \Var(f) \tag{SI}\\
            \sum_i \Var(\E_i[f]]) \le C \sum_i \E[\Var_i(f)] \tag{GE? (Appeared in Marton's paper)}
        \end{align*}

    Hypercontractivity under the simultaneous heat flow: 
    \href{https://arxiv.org/pdf/2507.18021}{Kook--Vempala}

    \href{https://arxiv.org/pdf/1911.07357}{Canonne--Chen--Kamath--Levi--Waingarten}, \href{https://arxiv.org/pdf/2302.09013}{Chen--Marcussen}.
